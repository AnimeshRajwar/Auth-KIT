Getting Started

Quick Start

AuthKit is a self-hosted auth service. Once running, your other apps authenticate users and verify tokens against it.

1
Install & configure
Copy the env file and add your secrets.
bash
npm install
cp .env.example .env
# Generate a strong secret:
node -e "console.log(require('crypto').randomBytes(64).toString('hex'))"
# Paste it into JWT_ACCESS_SECRET in .env
2
Start the server
The dashboard and API both run on the same port.
bash
npm start
# Server running at http://localhost:3001
# Dashboard UI at  http://localhost:3001/
3
Create a project & generate a key
Open the dashboard, register, create a project, and generate your first API key.
4
Use the key in your app
http
GET /your-protected-route HTTP/1.1
x-api-key: sk_a1b2c3d4e5f6...

Authentication

All protected endpoints accept two forms of authentication. You can use either in the same service.

Bearer Token (JWT)
Short-lived access token obtained from /auth/login. Best for user-facing apps where a session is managed.
Authorization: Bearer eyJhbGci...
API Key
Long-lived key generated from the dashboard. Best for server-to-server calls, CI pipelines, and scripts.
x-api-key: sk_a1b2c3d4e5f6...
⚠
Secret keys (sk_) have full API access — never expose them in client-side code. Use publishable keys (pk_) for browser or mobile apps.

Integration

Verify JWT in Your App

The fastest approach — verify the JWT locally without a network call. Share JWT_ACCESS_SECRET between AuthKit and your other services.

javascript
const jwt = require('jsonwebtoken');

function requireAuth(req, res, next) {
  const header = req.headers.authorization;
  if (!header) return res.status(401).json({ error: 'No token' });

  const token = header.split(' ')[1];
  try {
    req.user = jwt.verify(token, process.env.JWT_ACCESS_SECRET);
    // req.user = { id, email, iat, exp }
    next();
  } catch (err) {
    res.status(401).json({ error: 'Invalid or expired token' });
  }
}

// Usage
app.get('/profile', requireAuth, (req, res) => {
  res.json({ user: req.user });
});
python
import jwt
import os
from functools import wraps
from flask import request, jsonify, g

SECRET = os.environ['JWT_ACCESS_SECRET']

def require_auth(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        header = request.headers.get('Authorization', '')
        if not header.startswith('Bearer '):
            return jsonify(error='No token'), 401
        token = header[7:]
        try:
            g.user = jwt.decode(token, SECRET, algorithms=['HS256'])
        except jwt.PyJWTError:
            return jsonify(error='Invalid token'), 401
        return f(*args, **kwargs)
    return decorated

@app.route('/profile')
@require_auth
def profile():
    return jsonify(user=g.user)
go
import (
    "github.com/golang-jwt/jwt/v5"
    "net/http"
    "os"
    "strings"
)

func RequireAuth(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        auth := r.Header.Get("Authorization")
        if !strings.HasPrefix(auth, "Bearer ") {
            http.Error(w, "Unauthorized", 401); return
        }
        tokenStr := auth[7:]
        secret := []byte(os.Getenv("JWT_ACCESS_SECRET"))
        token, err := jwt.Parse(tokenStr, func(t *jwt.Token) (any, error) {
            return secret, nil
        })
        if err != nil || !token.Valid {
            http.Error(w, "Invalid token", 401); return
        }
        next.ServeHTTP(w, r)
    })
}

Validate an API Key

For API key validation, proxy the request to AuthKit's /auth/me endpoint. This is stateful — revoked keys are rejected instantly.

javascript
const AUTHKIT_URL = process.env.AUTHKIT_URL || 'http://localhost:3001';

async function validateApiKey(apiKey) {
  const res = await fetch(`${AUTHKIT_URL}/auth/me`, {
    headers: { 'x-api-key': apiKey },
  });
  if (!res.ok) throw new Error('Unauthorized');
  return res.json(); // { user: { id, email }, api_keys: [...] }
}

// Express middleware
async function requireApiKey(req, res, next) {
  const key = req.headers['x-api-key'];
  if (!key) return res.status(401).json({ error: 'API key required' });
  try {
    req.authData = await validateApiKey(key);
    next();
  } catch {
    res.status(401).json({ error: 'Invalid API key' });
  }
}
python
import os
import requests
from functools import wraps
from flask import request, jsonify, g

AUTHKIT_URL = os.environ.get('AUTHKIT_URL', 'http://localhost:3001')

def require_api_key(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        key = request.headers.get('x-api-key')
        if not key:
            return jsonify(error='API key required'), 401
        r = requests.get(f'{AUTHKIT_URL}/auth/me',
                         headers={'x-api-key': key}, timeout=5)
        if not r.ok:
            return jsonify(error='Invalid API key'), 401
        g.auth = r.json()
        return f(*args, **kwargs)
    return decorated
